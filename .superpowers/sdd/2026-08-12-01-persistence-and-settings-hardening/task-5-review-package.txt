+def _normalized_key(value: object) -> str:
+    return re.sub(r"[^a-z0-9]", "", str(value).casefold())
+
+
+def _load_or_create_salt(log_dir: Path) -> bytes:
+    salt_path = log_dir / ".runtime-salt"
+    try:
+        salt = salt_path.read_bytes()
+        if len(salt) >= 16:
+            return salt
+    except FileNotFoundError:
+        pass
+    salt = secrets.token_bytes(32)
+    try:
+        with salt_path.open("xb") as stream:
+            stream.write(salt)
+        try:
+            os.chmod(salt_path, 0o600)
+        except OSError:
+            pass
+        return salt
+    except FileExistsError:
+        existing = salt_path.read_bytes()
+        if len(existing) >= 16:
+            return existing
+        raise RuntimeError("runtime log salt is invalid")
+
+
+def _private_hash(value: object, salt: bytes) -> str:
+    payload = json.dumps(value, ensure_ascii=False, sort_keys=True, default=str).encode("utf-8")
+    return "sha256:" + hmac.new(salt, payload, hashlib.sha256).hexdigest()
+
+
+def _redact_value(value: object, *, salt: bytes | None, debug: bool) -> object:
+    if isinstance(value, Mapping):
+        cleaned: dict[str, object] = {}
+        for key, nested_value in value.items():
+            normalized = _normalized_key(key)
+            if "apikey" in normalized or "authorization" in normalized:
+                continue
+            if any(private_field in normalized for private_field in _PRIVATE_FIELDS):
+                if debug and salt is not None:
+                    cleaned[str(key)] = _private_hash(nested_value, salt)
+                continue
+            cleaned[str(key)] = _redact_value(nested_value, salt=salt, debug=debug)
+        return cleaned
+    if isinstance(value, (list, tuple)):
+        cleaned_items = []
+        for item in value:
+            if isinstance(item, (Mapping, list, tuple)):
+                cleaned_items.append(_redact_value(item, salt=salt, debug=debug))
+            elif debug and salt is not None:
+                cleaned_items.append(_private_hash(item, salt))
+        return cleaned_items
+    if value is None or isinstance(value, (str, int, float, bool)):
+        return value
+    return str(value)
+
+
+def redact_event(
+    event: Mapping[str, object],
+    *,
+    salt: bytes | None = None,
+    debug: bool | None = None,
+) -> dict[str, object]:
+    """Return only the non-content fields used for runtime auditing."""
+    debug_enabled = _SESSION_DEBUG if debug is None else bool(debug)
+    if debug_enabled:
+        if salt is None:
+            from app_paths import logs_dir
+
+            log_dir = logs_dir()
+            log_dir.mkdir(parents=True, exist_ok=True)
+            salt = _load_or_create_salt(log_dir)
+    redacted = {
+        key: _redact_value(event[key], salt=salt, debug=debug_enabled)
+        for key in sorted(_AUDIT_FIELDS)
+        if key in event

+        event_data = getattr(record, "event_data", {})
+        if not isinstance(event_data, Mapping):

+        redacted = redact_event(event_data, salt=self.salt)
+        redacted.setdefault("timestamp", datetime.now(timezone.utc).isoformat())
+        return json.dumps(redacted, ensure_ascii=False, sort_keys=True)

+    owned_handlers = [
