  387:     def strict_relevant_examples(self, conversation_text: str, limit: int = 6) -> list[dict[str, str]]:
  388:         query_tokens = _text_tokens(conversation_text)
  389:         if not query_tokens:
  390:             return []
  391:         last_other = _last_other_text(conversation_text)
  392:         last_tokens = _text_tokens(last_other)
  393:         scored = []
  394:         for index, item in enumerate(self.examples):
  395:             cue = str(item.get("cue", ""))
  396:             reply = str(item.get("reply", ""))
  397:             cue_tokens = _text_tokens(cue)
  398:             reply_tokens = _text_tokens(reply)
  399:             score = len(query_tokens & cue_tokens) * 3 + len(query_tokens & reply_tokens)
  400:             if last_tokens:
  401:                 score += len(last_tokens & cue_tokens) * 4
  402:             score += _phrasebook_intent_bonus(conversation_text, item)
  403:             if score:
  404:                 score += _example_priority(item) / 10
  405:                 scored.append((score, index, item))
  406:         scored.sort(key=lambda row: (row[0], row[1]), reverse=True)
  407:         selected = [item for _, _, item in scored[:limit]]
  408:         if len(selected) < limit:
  409:             seen = {item.get("conversation_hash", "") for item in selected}
  410:             for item in self.vector_relevant_examples(conversation_text, limit=limit * 2):
  411:                 key = item.get("conversation_hash", "")
  412:                 if key and key not in seen:
  413:                     selected.append(item)
  414:                     seen.add(key)
  415:                 if len(selected) >= limit:
  416:                     break
  417:         return selected[:limit]

  564:         query_tokens = _text_tokens(conversation_text)
  565:         query_vector = _text_vector(conversation_text)

  581:             if keyword_hits or reply_hits or vector_score >= 0.08:

  586:                 reasons.append("keyword")
  587:             if reply_hits:
  588:                 reasons.append("reply")
  589:             if vector_score >= 0.08:
  590:                 reasons.append("vector")
  591:             if _phrasebook_intent_bonus(conversation_text, item):
  592:                 reasons.append("intent")
  593:             if priority:
  594:                 reasons.append(f"priority:{priority}")
  595:             if not reasons:
  596:                 reasons.append("recent")

  598:             row.update(
  599:                 {
  600:                     "library_key": item.get("source", ""),
  601:                     "library_name": _library_name(str(item.get("source", ""))),

  603:                     "reasons": reasons,
  604:                     "priority": priority,
  605:                     "vector_score": round(vector_score, 4),

  969: def _enterprise_ratio_bonus(query_key: str, candidate_question_key: str) -> int:
  970:     """数字比例/关键数字短语加分。
  971: 
  972:     问法里的关键数字（如"1/3"归一化为"13"、"70周岁"的"70"）如果同时出现在
  973:     候选问题里，且数字前 1-4 个字的上下文有重叠，说明候选与问法在同一业务点
  974:     （如"负责人超理事数1/3" 应命中"负责人超过理事数的1/3"，而不是
  975:     "常务理事数的1/2"）。返回 0 或固定加分（15），避免并列误判。
  976:     """
  977:     if not query_key or not candidate_question_key:
  978:         return 0
  979:     query_nums = re.findall(r"\d+", query_key)
  980:     if not query_nums:
  981:         return 0
  982:     for num in query_nums:
  983:         if len(num) < 1 or num not in candidate_question_key:
  984:             continue
  985:         query_pos = query_key.find(num)
  986:         cand_pos = candidate_question_key.find(num)
  987:         query_ctx = query_key[max(0, query_pos - 4) : query_pos]
  988:         cand_ctx = candidate_question_key[max(0, cand_pos - 4) : cand_pos]
  989:         overlap = len(set(query_ctx) & set(cand_ctx))
  990:         if overlap >= 2:
  991:             return 15
  992:     return 0

## Fix Round 2

### Finding and root cause

- `_text_tokens()` only emits tokens of length two or greater, while `_text_vector()` previously emitted only 2-, 3-, and 4-character features.
- A valid authoritative example whose normalized text was a single character therefore produced `{}`. `_build_vector_items()` silently skipped it, allowing an empty `items` list to validate and incorrectly set `vector_ready=True`.
- `_text_vector()` now emits one deterministic fallback feature, `single:<normalized text>`, only when normalized text is non-empty and the normal feature set is empty. The same function is used for stored examples and queries, so one-character Chinese and ASCII replies remain mutually searchable. The vector validator was not relaxed.

### Tests added

- `test_single_character_examples_always_have_stable_vector_items[ascii]`
- `test_single_character_examples_always_have_stable_vector_items[chinese]`

Each case uses an empty cue and a one-character reply (`x` or `中`) and proves that the built item count equals the authoritative example count, the vector is non-empty, a rebuild finishes ready, and two subsequent loads do not trigger another vector rewrite.
